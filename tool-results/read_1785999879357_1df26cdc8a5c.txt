     5→/** Format amount in Bengali Taka */
     6→export function formatTaka(amount: number): string {
     7→  return `৳${amount.toLocaleString('en-IN')}`;
     8→}
     9→
    10→/** Fee frequency type */
    11→export type FeeFrequency = 'monthly' | 'quarterly' | 'annual' | 'one-time';
    12→
    13→/** Fee category */
    14→export interface FeeCategory {
    15→  id: string;
    16→  nameBn: string;
    17→  nameEn: string;
    18→  amount: number;
    19→  frequency: FeeFrequency;
    20→  studentCount: number;
    21→  icon: string;
    22→  isRecurring: boolean;
    23→}
    24→
    25→/** Fee structure cell (class × category) */
    26→export interface FeeStructureCell {
    27→  classId: string;
    28→  className: string;
    29→  categoryId: string;
    30→  amount: number;
    31→  isSet: boolean; // green if set, amber if default, gray if N/A
    32→}
    33→
    34→/** Invoice status */
    35→export type InvoiceStatus = 'paid' | 'partial' | 'overdue';
    36→
    37→/** Fee invoice */
    38→export interface FeeInvoice {
    39→  id: string;
    40→  invoiceNo: string;
    41→  studentId: string;
    42→  studentName: string;
    43→  studentNameBn: string;
    44→  className: string;
    45→  academicSession: string;
    46→  totalAmount: number;
    47→  paidAmount: number;
    48→  discountAmount: number;
    49→  balanceAmount: number;
    50→  status: InvoiceStatus;
    51→  dueDate: string;
    52→  generatedDate: string;
    53→  lineItems: InvoiceLineItem[];
    54→  payments: PaymentRecord[];
    55→}
    56→
    57→/** Invoice line item */
    58→export interface InvoiceLineItem {
    59→  categoryId: string;
    60→  categoryName: string;
    61→  amount: number;
    62→  discount: number;
    63→  netAmount: number;
    64→}
    65→
    66→/** Payment method */
    67→export type PaymentMethod = 'cash' | 'bkash' | 'bank' | 'cheque';
    68→
    69→/** Payment record */
    70→export interface PaymentRecord {
    71→  id: string;
    72→  invoiceId: string;
    73→  amount: number;
    74→  method: PaymentMethod;
    75→  date: string;
    76→  note: string;
    77→  receivedBy: string;
    78→}
    79→
    80→/** Recent collection record */
    81→export interface CollectionRecord {
    82→  id: string;
    83→  receiptNo: string;
    84→  studentName: string;
    85→  studentNameBn: string;
    86→  className: string;
    87→  amount: number;
    88→  method: PaymentMethod;
    89→  date: string;
    90→  invoiceNo: string;
    91→}
    92→
    93→/** Academic session */
    94→export interface AcademicSession {
    95→  id: string;
    96→  name: string;
    97→  isCurrent: boolean;
    98→}
    99→
   100→/** Class/Group for fee structure */
   101→export interface ClassGroup {
   102→  id: string;
   103→  name: string;
   104→  nameBn: string;
   105→  sections: string[];
   106→}
