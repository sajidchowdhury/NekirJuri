     1→// ============================================================
     2→// Madrasha ERP & Accounting Management System (SaaS)
     3→// Database Schema — Prisma (Dual Environment)
     4→// Dev: SQLite  |  Production: PostgreSQL
     5→// Version: 2.0
     6→// ============================================================
     7→// Environment Switch:
     8→//   Dev:  bun run db:dev       (switches to SQLite)
     9→//   Prod: bun run db:prod      (switches to PostgreSQL)
    10→//   The schema types (Decimal, Json) work with BOTH databases
    11→// ============================================================
    12→// Design Principles:
    13→//   - Multi-tenant: tenant_id on every business table
    14→//   - Composite unique constraints prevent cross-tenant data leaks
    15→//   - Soft delete (deleted_at) on transactional entities
    16→//   - Audit fields: created_at, updated_at, created_by, updated_by
    17→//   - Double-entry accounting foundation (chart_of_accounts, journal_entries)
    18→//   - Index strategy optimized for reporting queries
    19→//   - snake_case table names via @@map
    20→//   - Singular table names for consistency
    21→// ============================================================
    22→
    23→generator client {
    24→  provider = "prisma-client-js"
    25→}
    26→
    27→datasource db {
    28→  provider = "sqlite"
    29→  url      = env("DATABASE_URL")
    30→}

   125→model Subscription {
   126→  id              Int      @id @default(autoincrement())
   127→  tenantId        Int      @map("tenant_id")
   128→  planId          Int      @map("plan_id")
   129→  status          String   // trial | active | grace_period | restricted | suspended | terminated | cancelled
   130→  billingDuration Int      @default(1) @map("billing_duration") // 1, 6, or 12 months
   131→  paymentMethod   String?  @map("payment_method") // bkash | nagad | bank | manual
   132→  startDate       DateTime @map("start_date")
   133→  endDate         DateTime @map("end_date")

   135→  isAutoRenew     Boolean  @default(false) @map("is_auto_renew")
   136→  createdAt       DateTime @default(now()) @map("created_at")
   137→  updatedAt       DateTime @updatedAt @map("updated_at")
   138→
   139→  tenant   Tenant              @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   140→  plan     SubscriptionPlan    @relation(fields: [planId], references: [id])
   141→  payments SubscriptionPayment[]
   142→
   143→  @@index([tenantId])
   144→  @@index([status])
   145→  @@map("subscriptions")

   148→model SubscriptionPayment {
   149→  id             Int      @id @default(autoincrement())
   150→  subscriptionId Int      @map("subscription_id")
   151→  tenantId       Int      @map("tenant_id")
   152→  amount         Decimal
   153→  currency       String   @default("BDT")
   154→  paymentMethod  String   @map("payment_method") // bkash | nagad | bank | manual
   155→  paymentRef     String?  @map("payment_ref") // Transaction ID from bKash/Nagad
   156→  paymentPhone   String?  @map("payment_phone") // Phone number used for payment
   157→  status         String   // pending | verified | failed | refunded
   158→  billingPeriod  String   @map("billing_period") // e.g. "2025-01", "2025-H1", "2025"
   159→  duration       Int      // 1, 6, or 12 months
   160→  paidAt         DateTime? @map("paid_at")
   161→  verifiedAt     DateTime? @map("verified_at")
   162→  verifiedBy     Int?     @map("verified_by") // Admin who verified manual payment
   163→  notes          String?
   164→  createdAt      DateTime @default(now()) @map("created_at")
   165→  updatedAt      DateTime @updatedAt @map("updated_at")
   166→
   167→  subscription Subscription @relation(fields: [subscriptionId], references: [id], onDelete: Cascade)
   168→  tenant       Tenant       @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   169→
   170→  @@index([subscriptionId])
   171→  @@index([tenantId])
   172→  @@index([status])
   173→  @@index([paymentRef])
   174→  @@map("subscription_payments")
   175→}
